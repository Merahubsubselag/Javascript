=============================================================
   JAVASCRIPT: VAR, LET, CONST - INTERVIEW QUESTION BANK
=============================================================

--- [QUESTION 1: Hoisting & Temporal Dead Zone (TDZ)] ---
console.log(a);
console.log(b);

var a = 10;
let b = 20;

--- [QUESTION 2: Block Scope vs Function Scope] ---
let x = 10;
var y = 20;

if (true) {
  let x = 30;
  var y = 40;
}

console.log(x);
console.log(y);

--- [QUESTION 3: Re-declaration & Re-assignment] ---
var a = 1;
var a = 2; // Line 2

let b = 3;
let b = 4; // Line 5

const c = 5;
c = 6;     // Line 8


--- [QUESTION 4: Initialization Rule (Missing Value)] ---
let p;
p = 10;

const q;
q = 20;

--- [QUESTION 5: Valid Shadowing] ---
var score = 50;

{
  let score = 100;
  console.log("Block:", score);
}

console.log("Global:", score);

--- [QUESTION 6: Illegal Shadowing] ---
let count = 10;

{
  var count = 20;
  console.log("Inside:", count);
}

console.log("Outside:", count);

--- [QUESTION 7: Window / Global Object Attachment] ---
var greeting = "Namaste";
let farewell = "Alvida";

console.log(window.greeting);
console.log(window.farewell);

--- [QUESTION 8: Object Mutation with Const] ---
const user = {
  name: "Amit"
};

user.name = "Rahul";
console.log(user.name);

--- [QUESTION 9: Array Mutation with Const] ---
const numbers = [10, 20, 30];

numbers.push(40);
numbers[0] = 99;

console.log(numbers);

--- [QUESTION 10: TDZ Right-Hand Trap] ---
let x = x + 10;
console.log(x);

--- [QUESTION 11: Nested Blocks Scoping] ---
let a = 5;

{
  let a = 15;
  {
    let a = 25;
    console.log("Inner:", a);
  }
  console.log("Middle:", a);
}

console.log("Outer:", a);

--- [QUESTION 12: Scope Chain Fallback] ---
let a = 50;

{
  console.log("Inside:", a);
}

console.log("Outside:", a);

--- [QUESTION 13: Inner Block TDZ Trap] ---
let score = 100;

{
  console.log(score);
  let score = 200;
}

--- [QUESTION 14: Direct Assignment (No Keyword)] ---
name = "Rahul";
console.log(name);

--- [QUESTION 15: Mixed Scoping & Lifetime] ---
var a = 10;
let b = 20;

{
  var a = 100;
  let b = 200;
  const c = 300;
}

console.log(a);
console.log(b);
console.log(c);

let hoist to hota hai par Temporal Dead Zone (TDZ) mein rehta hai jab tak uski line execute na ho.

Reason:

let bahar wala safe rehta hai (20).
const block khatam hote hi memory se delete ho jata hai, isliye bahar c is not defined aata hai.

Lekin JS banane wale (Brendan Eich) ne ek alag soch ke sath ise design kiya tha. Aise samjho ki computer ke dimaag mein do alag-alag kaam hote hain:
1. Jagaah rokna (Memory Allocation): Jab computer code ko padhta hai, toh woh sirf itna dekhta hai ki use memory mein kitne dibbe (variables) banane hain. var x dekhte hi woh ek dibba bana deta hai.
2. Samaan dalna (Execution): Dibbe ke andar value (2) dalne ka kaam tabhi hota hai jab computer sach mein us line par pahunchta hai.
